# --- Transcription ---------------------------------------------------------
#
# Word-level timestamps are required for karaoke captions, and faster-whisper
# only returns them when word_timestamps=True is passed to transcribe().
#
# NOTE: do not "simplify" the audio loading in app/pipeline/transcribe.py back
# to faster_whisper.audio.decode_audio. Verified failure against this pin:
#
#   >>> av.open(path, mode="r", metadata_errors="ignore")
#   TypeError: open() got an unexpected keyword argument 'metadata_errors'
#
# decode_audio hardcodes that keyword, and PyAV removed it in v14. The local
# loader shells out to ffmpeg and reads 16 kHz mono PCM with the stdlib `wave`
# module instead, so the only audio dependency is an ffmpeg binary the pipeline
# already requires.
faster-whisper==1.2.1

# Pulled in transitively by faster-whisper and imported by it at module scope,
# so it cannot be treated as optional. Pinned explicitly for two reasons:
#
#   1. `av` is the package that dropped `metadata_errors` (see above). Leaving
#      it unpinned means the incompatible combination can reappear silently on
#      a fresh install, and the local loader's necessity becomes invisible.
#   2. On Windows the CUDA build of ctranslate2 is not bundled. 4.8.2 is the
#      verified version that ships a working CPU wheel; a newer resolution can
#      fail to install with no obvious cause.
#
# Verified CPU compute types for this ctranslate2: int8_float32, int8, float32.
av==19.0.1
ctranslate2==4.8.2
tokenizers==0.22.1

# numpy is a direct import in app/pipeline/transcribe.py, and ctranslate2
# requires it too. Bounded rather than pinned: 2.4.0 is the verified version,
# while an upper bound keeps a numpy 3.x release from breaking ctranslate2
# 4.8.2, which predates it.
#
# Note: `pip check` may report `numba ... requires numpy<2.4`. That comes from
# datashader, which is unrelated to this project, and does not affect the
# backend. See backend/README.md.
numpy>=1.26,<2.5

# --- Optional GPU ----------------------------------------------------------
#
# Not a hard requirement. app/pipeline/transcribe.py::resolve_device imports
# torch inside a try/except to probe for CUDA; with torch absent it returns
# "cpu" and the pipeline runs normally on CPU.
#
# torch is listed commented-out rather than pinned because it is a ~2.5GB
# install and there is no CUDA wheel for Windows. On Linux, or on Windows with
# an NVIDIA GPU, install the build matching your driver:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cu121
#
# On Windows, CUDA support requires a CUDA-enabled ctranslate2 build, which is
# not distributed through PyPI. CPU is the supported Windows path.
# torch==2.9.1
